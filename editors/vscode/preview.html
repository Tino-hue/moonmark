11/// A 2D point with generic coordinates
12#[derive(Debug, Show, Eq)]
13pub struct Point[T] {
14 x : T
15 y : T
16}
17
18impl Point[Int] {
19 /// Calculate distance from origin
20 pub fn distance(self) -> Double {
21 let dx = self.x.to_double()
22 let dy = self.y.to_double()
23 (dx * dx + dy * dy).sqrt()
24 }
25
26 pub fn translate(self, dx : Int, dy : Int) -> Point[Int] {
27 Point::new { x: self.x + dx, y: self.y + dy }
28 }
29}

54/// Async HTTP request handler
55async fn fetch_data(url : String) -> Result[String, Error] {
56 let (resp, body) = @http.get(url)
57 guard resp.code is (200..<300) else {
58 return Err(Error("HTTP " + resp.code.to_string()))
59 }
60
61 // String interpolation example
62 let msg = $"Fetched \{body.length()} bytes from \{url}"
63
64 // Raw multi-line string
65 let html = #|
66<div class="result">
67 <p>\{msg}</p>
68</div>
69|#
70
71 Ok(html)
72}

74// Pattern matching showcase
75fn process_color(c : Color) -> String {
76 match c {
77 Red => "#FF0000"
78 Green => "#00FF00"
79 Blue => "#0000FF"
80 Rgb(r, g, b) => $"rgb(\{r},\{g},\{b})"
81 Hex(s) => s
82 }
83}

85// Guard expression
86fn unwrap[T](opt : Option[T], default : T) -> T {

91// For loop with range
